/* =========================================================
   STORAGE
========================================================= */

function loadState() {

  try {

    const raw =
      localStorage.getItem(
        STORAGE_KEY
      );

    if (!raw) {

      return structuredClone(
        defaultState
      );
    }

    const parsed =
      JSON.parse(raw);

    return {

      ...structuredClone(
        defaultState
      ),

      ...parsed,

      meta: {

        ...defaultState.meta,

        ...(parsed.meta || {})
      },

      trees:
        Array.isArray(parsed.trees)
          ? parsed.trees
          : [],

      boundary:
        Array.isArray(parsed.boundary)
          ? parsed.boundary
          : [],

      plots:
        Array.isArray(parsed.plots)
          ? parsed.plots
          : []
    };

  }

  catch(error) {

    console.error(error);

    return structuredClone(
      defaultState
    );
  }
}


function saveState(
  message = "Saved"
) {

  try {

    localStorage.setItem(

      STORAGE_KEY,

      JSON.stringify(state)
    );

    $("saveStatus")
      .textContent =
      message;

  }

  catch(error) {

    console.error(error);

    alert(
      "Storage is full. Export JSON backup and remove some photos/data."
    );
  }
}

/* =========================================================

========================================================= */

function nowISO() {

  return new Date()
    .toISOString();
}


function fmt(
  number,
  digits = 6
) {

  return Number.isFinite(
    Number(number)
  )

    ? Number(number)
        .toFixed(digits)

    : "—";
}


function escapeHtml(value) {

  return String(
    value ?? ""
  )

  .replaceAll(
    "&",
    "&amp;"
  )

  .replaceAll(
    "<",
    "&lt;"
  )

  .replaceAll(
    ">",
    "&gt;"
  )

  .replaceAll(
    '"',
    "&quot;"
  )

  .replaceAll(
    "'",
    "&#039;"
  );
}


function safeFileName(name) {

  return String(
    name ||
    "forest-survey"
  )

  .replace(
    /[^a-z0-9_\-]+/gi,
    "_"
  )

  .replace(
    /^_+|_+$/g,
    ""
  )

  .slice(
    0,
    80
  )

  ||
  "forest-survey";
}

/* =========================================================
   GPS ACCURACY
========================================================= */

function gpsQuality(
  accuracy
) {

  if (!Number.isFinite(
    accuracy
  ))

    return "Waiting for GPS";

  if (accuracy <= 3)
    return "Excellent (reported)";

  if (accuracy <= 5)
    return "Very Good (reported)";

  if (accuracy <= 10)
    return "Good (reported)";

  if (accuracy <= 20)
    return "Moderate (reported)";

  if (accuracy <= 50)
    return "Weak (reported)";

  return "Poor (reported)";
}


/* =========================================================
   GPS SUCCESS
========================================================= */

function onGpsSuccess(
  position
) {

  const c =
    position.coords;

  gps.current = {

    lat:
      c.latitude,

    lng:
      c.longitude,

    alt:
      c.altitude,

    accuracy:
      c.accuracy,

    heading:
      c.heading,

    speed:
      c.speed,

    timestamp:
      position.timestamp ||
      Date.now()
  };


  updateGpsReadout();

  updateGpsMap();

  updateDashboard();
}


/* =========================================================
   GPS ERROR
========================================================= */

function onGpsError(
  error
) {

  let message =
    "GPS error.";

  if (error.code === 1) {

    message =
      "Location permission denied. Enable location permission.";
  }

  if (error.code === 2) {

    message =
      "Location unavailable. Move to an open area.";
  }

  if (error.code === 3) {

    message =
      "GPS timeout. Wait and try again.";
  }

  $("gpsMessage")
    .textContent =
    message;

  $("gpsMessage")
    .className =
    "message warning";
}


/* =========================================================
   START GPS
========================================================= */

function startGPS() {

  if (!navigator.geolocation) {

    alert(
      "This browser does not support GPS."
    );

    return;
  }


  if (
    gps.watchId !== null
  ) {

    return;
  }


  $("gpsMessage")
    .textContent =
    "Starting high accuracy GPS...";

  $("gpsMessage")
    .className =
    "message info";


  gps.watchId =
    navigator.geolocation
      .watchPosition(

        onGpsSuccess,

        onGpsError,

        {

          enableHighAccuracy:
            true,

          maximumAge:
            1000,

          timeout:
            15000
        }
      );
}


/* =========================================================
   STOP GPS
========================================================= */

function stopGPS() {

  if (
    gps.watchId !== null
  ) {

    navigator.geolocation
      .clearWatch(
        gps.watchId
      );

    gps.watchId =
      null;
  }

  $("gpsMessage")
    .textContent =
    "GPS stopped.";

  $("gpsMessage")
    .className =
    "message info";
}

/* =========================================================
   GPS MAP
========================================================= */

function updateGpsMap() {

  if (!gps.current)
    return;

  const p =
    gps.current;


  if (!gps.marker) {

    gps.marker =
      L.marker(

        [
          p.lat,
          p.lng
        ],

        {
          icon:
            gpsIcon
        }

      )

      .bindPopup(
        "📡 Current GPS Position"
      )

      .addTo(
        gpsLayer
      );

  }

  else {

    gps.marker
      .setLatLng(
        [
          p.lat,
          p.lng
        ]
      );
  }


  if (!gps.accuracyCircle) {

    gps.accuracyCircle =
      L.circle(

        [
          p.lat,
          p.lng
        ],

        {

          radius:
            Math.max(
              1,
              p.accuracy || 1
            ),

          color:
            "#1976d2",

          fillColor:
            "#1976d2",

          fillOpacity:
            0.10,

          weight:
            1
        }

      )

      .addTo(
        gpsLayer
      );

  }

  else {

    gps.accuracyCircle
      .setLatLng(
        [
          p.lat,
          p.lng
        ]
      );

    gps.accuracyCircle
      .setRadius(
        Math.max(
          1,
          p.accuracy || 1
        )
      );
  }
}

/* =========================================================
   DISTANCE
========================================================= */

function distanceMeters(
  a,
  b
) {

  const R =
    6371008.8;

  const lat1 =
    a.lat *
    Math.PI / 180;

  const lat2 =
    b.lat *
    Math.PI / 180;

  const dLat =
    (b.lat - a.lat) *
    Math.PI / 180;

  const dLng =
    (b.lng - a.lng) *
    Math.PI / 180;


  const h =

    Math.sin(
      dLat / 2
    ) ** 2

    +

    Math.cos(lat1) *
    Math.cos(lat2) *

    Math.sin(
      dLng / 2
    ) ** 2;


  return (

    2 * R *

    Math.asin(

      Math.min(
        1,
        Math.sqrt(h)
      )

    )

  );
}


/* =========================================================
   PERIMETER
========================================================= */

function polygonPerimeter(
  points
) {

  if (
    points.length < 2
  )

    return 0;


  let total = 0;


  for (
    let i = 1;
    i < points.length;
    i++
  ) {

    total +=
      distanceMeters(
        points[i - 1],
        points[i]
      );
  }


  if (
    state.boundaryClosed &&
    points.length >= 3
  ) {

    total +=
      distanceMeters(

        points[
          points.length - 1
        ],

        points[0]
      );
  }


  return total;
}


/* =========================================================
   AREA
========================================================= */

function polygonAreaM2(
  points
) {

  if (
    points.length < 3
  )

    return 0;


  const lat0 =

    points.reduce(
      (sum, p) =>
        sum + p.lat,
      0
    )

    /

    points.length;


  const R =
    6371008.8;

  const rad =
    Math.PI / 180;


  const xy =
    points.map(
      p => ({

        x:
          R *
          p.lng *
          rad *
          Math.cos(
            lat0 * rad
          ),

        y:
          R *
          p.lat *
          rad
      })
    );


  let area = 0;


  for (
    let i = 0;
    i < xy.length;
    i++
  ) {

    const j =
      (i + 1)
      % xy.length;


    area +=

      xy[i].x *
      xy[j].y

      -

      xy[j].x *
      xy[i].y;
  }


  return Math.abs(
    area
  ) / 2;
}


/* =========================================================
   DASHBOARD
========================================================= */

function updateDashboard() {

  const acc =
    gps.current?.accuracy;


  $("gpsAccuracy")
    .textContent =

    Number.isFinite(acc)

      ? `${acc.toFixed(1)} m`

      : "—";


  $("gpsQuality")
    .textContent =
    gpsQuality(acc);


  $("treeCount")
    .textContent =
    state.trees.length;


  $("boundaryCount")
    .textContent =
    state.boundary.length;


  const area =
    state.boundaryClosed

      ? polygonAreaM2(
          state.boundary
        )

      : 0;


  const perimeter =
    polygonPerimeter(
      state.boundary
    );


  $("areaValue")
    .textContent =
    `${area.toFixed(2)} m²`;


  $("areaHectare")
    .textContent =
    `${(
      area / 10000
    ).toFixed(4)} ha`;


  $("perimeterValue")
    .textContent =
    `${perimeter.toFixed(2)} m`;
}


/* =========================================================
   BOUNDARY RENDER
========================================================= */

function renderBoundary() {

  boundaryLayer.clearLayers();


  const points =
    state.boundary
      .map(
        p => [
          p.lat,
          p.lng
        ]
      );


  state.boundary.forEach(
    (p, i) => {

      const marker =
        L.circleMarker(

          [
            p.lat,
            p.lng
          ],

          {

            radius:
              5,

            color:
              "#ff8f00",

            fillColor:
              "#ffb300",

            fillOpacity:
              0.9,

            weight:
              2
          }
        );


      marker
        .bindPopup(

          `<b>Boundary Point ${
            i + 1
          }</b><br>

          Lat:
          ${fmt(p.lat,7)}<br>

          Lng:
          ${fmt(p.lng,7)}<br>

          Accuracy:
          ${fmt(p.accuracy,2)} m<br>

          Source:
          ${escapeHtml(
            p.source
          )}`

        )

        .addTo(
          boundaryLayer
        );
    }
  );


  if (
    points.length >= 2
  ) {

    L.polyline(

      points,

      {

        color:
          "#ff8f00",

        weight:
          4
      }

    ).addTo(
      boundaryLayer
    );
  }


  if (
    state.boundaryClosed &&
    points.length >= 3
  ) {

    L.polygon(

      points,

      {

        color:
          "#2e7d32",

        fillColor:
          "#4caf50",

        fillOpacity:
          0.18,

        weight:
          3
      }

    ).addTo(
      boundaryLayer
    );
  }


  renderBoundaryTable();

  updateDashboard();
}

/* =========================================================
   TREE CALCULATION
========================================================= */

function basalAreaFromDbh(
  dbhCm
) {

  if (
    !Number.isFinite(dbhCm) ||
    dbhCm <= 0
  )

    return 0;


  return (

    Math.PI *

    Math.pow(
      dbhCm / 200,
      2
    )

  );
}


function volumeFromValues(
  dbhCm,
  heightM,
  formFactor
) {

  const ba =
    basalAreaFromDbh(
      dbhCm
    );


  if (
    !Number.isFinite(heightM) ||
    heightM <= 0
  )

    return 0;


  const ff =
    Number.isFinite(
      formFactor
    )

      ? formFactor

      : 0.45;


  return (
    ba *
    heightM *
    ff
  );
}

/* =========================================================
   NEXT TREE ID
========================================================= */

function nextTreeId() {

  const used =
    new Set(
      state.trees
        .map(
          t => t.id
        )
    );


  let n = 1;


  while (
    used.has(
      `T-${String(n)
        .padStart(3,"0")}`
    )
  ) {

    n++;
  }


  return (
    `T-${String(n)
      .padStart(3,"0")}`
  );
}


/* =========================================================
   PHOTO COMPRESSION
========================================================= */

function compressImage(
  file,
  maxSize = 900,
  quality = 0.68
) {

  return new Promise(
    (
      resolve,
      reject
    ) => {

      if (!file) {

        resolve(null);

        return;
      }


      const reader =
        new FileReader();


      reader.onload = () => {

        const img =
          new Image();


        img.onload = () => {

          const scale =
            Math.min(

              1,

              maxSize /
              Math.max(
                img.width,
                img.height
              )

            );


          const canvas =
            document.createElement(
              "canvas"
            );


          canvas.width =
            Math.round(
              img.width *
              scale
            );


          canvas.height =
            Math.round(
              img.height *
              scale
            );


          const ctx =
            canvas.getContext(
              "2d"
            );


          ctx.drawImage(

            img,

            0,
            0,

            canvas.width,
            canvas.height
          );


          resolve(

            canvas.toDataURL(
              "image/jpeg",
              quality
            )

          );

        };


        img.onerror =
          reject;


        img.src =
          reader.result;
      };


      reader.onerror =
        reject;


      reader.readAsDataURL(
        file
      );
    }
  );
}


/* =========================================================
   MARK TREE
========================================================= */

async function markTree() {

  if (!gps.current) {

    alert(
      "GPS position is not available. Start GPS first."
    );

    return;
  }


  const id =
    $("treeId")
      .value
      .trim()
      ||
      nextTreeId();


  const species =
    $("treeSpecies")
      .value
      .trim()
      ||
      "Unknown";


  const plotId =
    $("treePlotId")
      .value
      .trim();


  const dbh =
    Number(
      $("treeDbh").value
    );


  const height =
    Number(
      $("treeHeight").value
    );


  const formFactor =
    Number(
      $("treeFormFactor").value
    )
    ||
    0.45;


  const condition =
    $("treeCondition")
      .value;


  if (
    state.trees.some(
      t => t.id === id
    )
  ) {

    alert(
      `Tree ID ${id} already exists.`
    );

    return;
  }


  let photo =
    null;


  const file =
    $("treePhoto")
      .files[0];


  if (file) {

    try {

      photo =
        await compressImage(
          file
        );

    }

    catch(error) {

      console.error(error);

      alert(
        "Photo could not be processed."
      );
    }
  }


  const ba =
    basalAreaFromDbh(
      dbh
    );


  const volume =
    volumeFromValues(
      dbh,
      height,
      formFactor
    );


  const tree = {

    id,

    plotId,

    species,

    dbh:
      Number.isFinite(dbh)
        ? dbh
        : null,

    height:
      Number.isFinite(height)
        ? height
        : null,

    formFactor,

    basalArea:
      ba,

    volume,

    condition,

    photo,

    lat:
      gps.current.lat,

    lng:
      gps.current.lng,

    alt:
      gps.current.alt,

    accuracy:
      gps.current.accuracy,

    timestamp:
      nowISO()
  };


  state.trees.push(
    tree
  );


  saveState(
    `Tree ${id} saved`
  );


  $("treeId")
    .value =
    nextTreeId();


  $("treePhoto")
    .value =
    "";


  renderTrees();

  renderMapTrees();

  updateDashboard();


  map.setView(

    [
      tree.lat,
      tree.lng
    ],

    Math.max(
      map.getZoom(),
      18
    )
  );
}

    return;


  map.setView(

    [

      tree.lng
    ],

    19
  );


  treeLayer.eachLayer(
    layer => {

      if (
        layer.getLatLng &&
        Math.abs(
          layer.getLatLng().lat -
          tree.lat
        ) < 0.00000001 &&

        Math.abs(
          layer.getLatLng().lng -
          tree.lng
        ) < 0.00000001
      ) {

        layer.openPopup();
      }
    }
  );
};

/* =========================================================
   EXPORT KMZ
========================================================= */

function dataUrlToUint8Array(
  dataUrl
) {

  const base64 =
    dataUrl.split(",")[1];


  const binary =
    atob(base64);


  const bytes =
    new Uint8Array(
      binary.length
    );


  for (
    let i = 0;
    i < binary.length;
    i++
  ) {

    bytes[i] =
      binary.charCodeAt(i);
  }


  return bytes;
}


async function exportKmz() {

  syncMetaFromForm();


  if (
    typeof JSZip ===
    "undefined"
  ) {

    alert(
      "JSZip did not load. Check internet connection."
    );

    return;
  }


  const zip =
    new JSZip();


  const kml =
    buildKml();


  zip.file(
    "doc.kml",
    kml
  );


  /* ADD PHOTOS */

  state.trees.forEach(
    t => {

      if (!t.photo)
        return;


      const filename =
        `images/${safeFileName(
          t.id
        )}.jpg`;


      zip.file(

        filename,

        dataUrlToUint8Array(
          t.photo
        )

      );
    }
  );


  const blob =
    await zip.generateAsync({

      type:
        "blob",

      compression:
        "DEFLATE",

      compressionOptions: {

        level:
          6
      }

    });


  downloadBlob(

    blob,

    `${safeFileName(
      state.meta.projectName
    )}_forest_survey.kmz`

  );
}


/* =========================================================
   GEOJSON
========================================================= */

function exportGeoJson() {

  syncMetaFromForm();


  const features = [];


  /* BOUNDARY POINTS */

  state.boundary.forEach(
    (p, i) => {

      features.push({

        type:
          "Feature",

        properties: {

          type:
            "boundary_point",

          point_no:
            i + 1,

          accuracy_m:
            p.accuracy,

          source:
            p.source

        },

        geometry: {

          type:
            "Point",

          coordinates: [

            p.lng,

            p.lat,

            Number.isFinite(p.alt)
              ? p.alt
              : 0

          ]

        }

      });
    }
  );


  /* POLYGON */

  if (

    state.boundaryClosed &&

    state.boundary.length >= 3

  ) {

    const coordinates =

      state.boundary.map(

        p => [

          p.lng,

          p.lat,

          Number.isFinite(p.alt)
            ? p.alt
            : 0

        ]

      );


    coordinates.push(
      coordinates[0]
    );


    features.push({

      type:
        "Feature",

      properties: {

        type:
          "forest_boundary",

        project:
          state.meta.projectName,

        area_m2:
          polygonAreaM2(
            state.boundary
          ),

        area_ha:
          polygonAreaM2(
            state.boundary
          ) / 10000,

        perimeter_m:
          polygonPerimeter(
            state.boundary
          )

      },

      geometry: {

        type:
          "Polygon",

        coordinates:
          [coordinates]

      }

    });
  }


  /* TREES */

  state.trees.forEach(
    t => {

      features.push({

        type:
          "Feature",

        properties: {

          type:
            "tree",

          tree_id:
            t.id,

          plot_id:
            t.plotId,

          species:
            t.species,

          dbh_cm:
            t.dbh,

          height_m:
            t.height,

          form_factor:
            t.formFactor,

          basal_area_m2:
            t.basalArea,

          volume_m3:
            t.volume,

          condition:
            t.condition,

          gps_accuracy_m:
            t.accuracy,

          timestamp:
            t.timestamp

        },

        geometry: {

          type:
            "Point",

          coordinates: [

            t.lng,

            t.lat,

            Number.isFinite(t.alt)
              ? t.alt
              : 0

          ]

        }

      });
    }
  );


  /* PLOTS */

  state.plots.forEach(
    p => {

      features.push({

        type:
          "Feature",

        properties: {

          type:
            "plot_center",

          plot_id:
            p.id,

          plot_type:
            p.type,

          radius_m:
            p.radius,

          gps_accuracy_m:
            p.accuracy

        },

        geometry: {

          type:
            "Point",

          coordinates: [

            p.lng,

            p.lat,

            Number.isFinite(p.alt)
              ? p.alt
              : 0

          ]

        }

      });
    }
  );


  const geojson = {

    type:
      "FeatureCollection",

    name:
      state.meta.projectName,

    features

  };


  downloadBlob(

    new Blob(

      [
        JSON.stringify(
          geojson,
          null,
          2
        )
      ],

      {
        type:
          "application/geo+json"
      }

    ),

    `${safeFileName(
      state.meta.projectName
    )}_forest_survey.geojson`

  );
}

/* =========================================================
   IMPORT JSON
========================================================= */

function importJson(
  file
) {

  const reader =
    new FileReader();


  reader.onload = () => {

    try {

      const obj =
        JSON.parse(
          reader.result
        );


      const incoming =
        obj.state ||
        obj;


      if (

        !incoming ||

        !Array.isArray(
          incoming.trees
        ) ||

        !Array.isArray(
          incoming.boundary
        ) ||

        !Array.isArray(
          incoming.plots
        )

      ) {

        throw new Error(
          "Invalid backup"
        );
      }


      state = {

        ...structuredClone(
          defaultState
        ),

        ...incoming,

        meta: {

          ...defaultState.meta,

          ...(incoming.meta || {})

        }

      };


      saveState(
        "Backup imported"
      );


      syncMetaToForm();

      renderAll();


      alert(
        "Backup imported successfully."
      );

    }

    catch(error) {

      console.error(error);

      alert(
        "Invalid JSON backup file."
      );
    }
  };


  reader.readAsText(
    file
  );
}
